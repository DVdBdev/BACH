% Normal sectioning remains available, with modest sizes matching the template.
\titleformat{\section}
  {\Large\bfseries\color{PrimaryColor}}{\thesection}{0.7em}{}
\titlespacing*{\section}{0pt}{2.1ex plus 0.5ex minus 0.2ex}{0.7ex}
\titleformat{\subsection}
  {\large\bfseries\color{PrimaryColor}}{\thesubsection}{0.7em}{}
\titlespacing*{\subsection}{0pt}{1.7ex plus 0.4ex minus 0.2ex}{0.5ex}
